<script>


const VOWELS=['a','e','i','o','u']

const CONSONANTS='bcdfghjklmnpqrstvwxyz'.split('')

const RANKS=[

[0,'Beginner'],[0.04,'Nice'],[0.12,'Solid'],

[0.25,'Good'],[0.45,'Great'],[0.65,'Amazing'],[0.90,'Genius']

]


let wordSet=[]

let letters=[]

let keyLetter=''

let valid=[]

let found=[]

let score=0

let input=''


function shuffle(a){

return a.sort(()=>Math.random()-0.5)

}


function flash(t){

document.getElementById("msg").textContent=t

setTimeout(()=>document.getElementById("msg").textContent="",1800)

}


async function loadWords(){


const res=await fetch("https://raw.githubusercontent.com/dwyl/english-words/master/words_alpha.txt")

const text=await res.text()


wordSet=text.split("\n")

.map(w=>w.trim().toLowerCase())

.filter(w=>{

if(w.length<4||w.length>10)return false

if(!/^[a-z]+$/.test(w))return false

return new Set(w).size<=7

})


newGame()


}


function newGame(){


let best=null

let bestN=0


for(let attempt=0;attempt<80 && bestN<30;attempt++){


const nv=2+Math.floor(Math.random()*2)

const ls=[...shuffle([...VOWELS]).slice(0,nv),...shuffle([...CONSONANTS]).slice(0,7-nv)]

const k=ls[Math.floor(Math.random()*7)]


const lset=new Set(ls)

const v=[]


for(const w of wordSet){


let ok=true

let hasK=false


for(const c of w){

if(!lset.has(c)){ok=false;break}

if(c===k)hasK=true

}


if(ok && hasK)v.push(w)


}


if(v.length>bestN){

bestN=v.length

best={letters:ls,key:k,valid:v}

}


}


letters=best.letters

keyLetter=best.key

valid=best.valid

found=[]

score=0

input=''


render()


}


function render(){


const grid=document.getElementById("grid")

grid.innerHTML=""


const ring=letters.filter(l=>l!==keyLetter)


const layout=[ring[0],ring[1],ring[2],ring[3],keyLetter,ring[4],ring[5]]


layout.forEach((l,i)=>{


const d=document.createElement("div")

d.className="cell"+(l===keyLetter?" center":"")

d.textContent=l.toUpperCase()


d.onclick=()=>addLetter(l)


grid.appendChild(d)


})


document.getElementById("input").textContent=input.toUpperCase()


document.getElementById("score").textContent=score

document.getElementById("found").textContent=found.length

document.getElementById("total").textContent=valid.length


const maxScore=valid.reduce((s,w)=>s+(w.length===4?1:w.length),0)

const pct=maxScore?score/maxScore:0


document.getElementById("bar").style.width=Math.min(pct*100,100)+"%"


const rank=RANKS.reduce((r,[t,n])=>pct>=t?n:r,"Beginner")

document.getElementById("rank").textContent=rank.toUpperCase()


const words=document.getElementById("words")

words.innerHTML=""


found.forEach(w=>{

const t=document.createElement("span")

t.className="tag"

t.textContent=w.toUpperCase()

words.appendChild(t)

})


}


function addLetter(l){

input+=l

render()

}


function submit(){


const w=input.toLowerCase().trim()


if(w.length<4){flash("4 letters minimum");return}

if(!w.includes(keyLetter)){flash("Must include "+keyLetter.toUpperCase());return}

if([...w].some(c=>!letters.includes(c))){flash("Invalid letters");return}

if(found.includes(w)){flash("Already found");return}

if(!valid.includes(w)){flash("Not a word");return}


const pts=w.length===4?1:w.length


found.push(w)

found.sort()


score+=pts

input=""


flash("+"+pts)


render()


}


function shuffleLetters(){


const ring=shuffle(letters.filter(l=>l!==keyLetter))

letters=[...ring,keyLetter]

render()


}


document.addEventListener("keydown",e=>{


if(e.key==="Enter")submit()


else if(e.key==="Backspace"){

input=input.slice(0,-1)

render()

}


else if(/^[a-zA-Z]$/.test(e.key)){

input+=e.key.toLowerCase()

render()

}


})


loadWords()


</script>
